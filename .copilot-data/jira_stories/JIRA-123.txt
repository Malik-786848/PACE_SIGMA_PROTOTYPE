    def _discover_project_folders(self) -> List[Path]:
        """Discover all project folders in the base folder"""
        folders = []
        
        # Check if base_folder itself is a project
        if (self.base_folder / "jira_stories").exists() or \
           (self.base_folder / ".copilot-data").exists():
            folders.append(self.base_folder)
        
        # Look for subdirectories that are projects
        if self.base_folder.exists():
            for item in sorted(self.base_folder.iterdir()):
                if item.is_dir() and not item.name.startswith('.'):
                    if (item / "jira_stories").exists() or \
                       (item / ".copilot-data").exists() or \
                       (item / "open_prompts").exists():
                        if item not in folders:
                            folders.append(item)
        
        return folders or [self.base_folder]

    def _start_monitor(self):
        """Start monitoring"""
        if self.monitoring:
            messagebox.showwarning("Already Running", "Monitor is already running!")
            return
        
        try:
            # Start monitor in separate thread
            def run_monitor():
                try:
                    script_path = self.project_folder / "auto_watch.py"
                    self.monitor_process = subprocess.Popen(
                        [sys.executable, str(script_path)],
                        cwd=str(self.project_folder),
                        stdout=subprocess.PIPE,
                        stderr=subprocess.PIPE,
                        text=True
                    )
                except Exception as e:
                    messagebox.showerror("Error", f"Failed to start monitor: {e}")
                    self.monitoring = False
                    self._update_status()
            
            self.monitoring = True
            self._update_status()
            
            thread = threading.Thread(target=run_monitor, daemon=True)
            thread.start()
            
            self.start_btn.config(state=tk.DISABLED)
            self.stop_btn.config(state=tk.NORMAL)
            
            messagebox.showinfo(
                "Monitor Started",
                "✓ Monitor is running!\n\n"
                "Now open VS Code and chat with Copilot.\n"
                "Chats will be captured automatically.\n\n"
                "Click 'Refresh Stats' to see results."
            )
            
            # Auto-refresh logs
            self._refresh_log()
            
        except Exception as e:
            messagebox.showerror("Error", f"Failed to start monitor: {e}")
            self.monitoring = False
            self._update_status()
    
    def _stop_monitor(self):
        """Stop monitoring"""
        if not self.monitoring:
            messagebox.showwarning("Not Running", "Monitor is not running!")
            return
        
        try:
            if self.monitor_process:
                self.monitor_process.terminate()
                self.monitor_process.wait(timeout=5)
            
            self.monitoring = False
            self._update_status()
            
            self.start_btn.config(state=tk.NORMAL)
            self.stop_btn.config(state=tk.DISABLED)
            
            messagebox.showinfo(
                "Monitor Stopped",
                "✓ Monitor has been stopped.\n\n"
                "Click 'Start Monitor' to resume."
            )
        except Exception as e:
            messagebox.showerror("Error", f"Failed to stop monitor: {e}")
    
    def _update_status(self):

        if self.monitoring:

                text="● RUNNING",
                foreground="green"

        else:
            self.status_label.config(

            )
            self.status_detail.config(

            )
    
    def _on_project_changed(self):
        """Handle project selection change"""
        project_name = self.project_var.get()
        self.current_project = next(
            (p for p in self.project_folders if p.name == project_name),
            self.project_folders[0]
        )
        self._refresh_stats()
    
    def _refresh_stats(self):
        """Refresh statistics"""
        try:
            jira_dir = self.current_project / "jira_stories"
            open_dir = self.current_project / "open_prompts"
            
            jira_count = len(list(jira_dir.glob("*.txt"))) if jira_dir.exists() else 0
            open_count = len(list(open_dir.glob("*.txt"))) if open_dir.exists() else 0
            total_count = jira_count + open_count
            
            # Calculate tokens and cost
            total_tokens = 0
            total_cost = 0.0
            
            for txt_file in list(jira_dir.glob("*.txt")) + list(open_dir.glob("*.txt")):
                try:
                    content = txt_file.read_text()
                    # Extract tokens from file
                    for line in content.split('\n'):
                        if 'Total Tokens:' in line:
                            try:
                                tokens = int(line.split(':')[1].strip())
                                total_tokens += tokens
                            except:
                                pass
                        if 'Cost: $' in line:
                            try:
                                cost = float(line.split('$')[1].strip())
                                total_cost += cost
                            except:
                                pass
                except:
                    pass
            
            self.stat_total.config(text=str(total_count))
            self.stat_jira.config(text=str(jira_count))
            self.stat_open.config(text=str(open_count))
            self.stat_tokens.config(text=str(total_tokens))
            self.stat_cost.config(text=f"${total_cost:.4f}")
            
            self._refresh_jira()
            self._refresh_open()
            
        except Exception as e:
            messagebox.showerror("Error", f"Failed to refresh stats: {e}")

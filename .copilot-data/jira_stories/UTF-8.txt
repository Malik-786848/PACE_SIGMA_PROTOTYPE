def estimate_tokens(text):
    """Estimate token count from text (~1 token per 4 characters)."""
    if not text:
        return 0
    return max(1, len(text) // 4)

def calculate_cost(input_tokens, output_tokens, pricing_dict):
    """Calculate cost in AICs based on REAL pricing from Copilot."""
    if not pricing_dict or pricing_dict.get("input") == 0:
        return 0.0
    
    input_cost = (input_tokens / 1_000_000) * pricing_dict.get("input", 0)
    output_cost = (output_tokens / 1_000_000) * pricing_dict.get("output", 0)
    return input_cost + output_cost

def extract_tokens_and_classify(workspace_hash):
    """Extract tokens and classify by Jira story or open prompt."""
    session_dir = os.path.join(WORKSPACE_STORAGE, workspace_hash, "chatSessions")
    if not os.path.isdir(session_dir):
        return defaultdict(lambda: {"input": 0, "output": 0, "chats": 0}), {"input": 0, "output": 0, "chats": 0}
    
    story_tokens = defaultdict(lambda: {"input": 0, "output": 0, "chats": 0})
    open_tokens = {"input": 0, "output": 0, "chats": 0}
    
    for path in glob.glob(os.path.join(session_dir, "*.json")) + glob.glob(os.path.join(session_dir, "*.jsonl")):
        try:
            requests = _load_session_requests(path)
            for req in requests:
                prompt_text = _text_value((req.get("message") or {}).get("text", ""))
                response_parts = req.get("response") or []
                response_text = ""
                for part in response_parts:
                    if isinstance(part, dict):
                        response_text += _text_value(part.get("value", ""))
                    elif isinstance(part, str):
                        response_text += part
                
                if not prompt_text.strip():
                    continue
                
                input_tokens = estimate_tokens(prompt_text)
                output_tokens = estimate_tokens(response_text)
                
                # Get Jira keys from prompt
                keys = get_jira_keys(prompt_text)
                
                if keys:
                    for key in set(keys):
                        story_tokens[key]["input"] += input_tokens
                        story_tokens[key]["output"] += output_tokens
                        story_tokens[key]["chats"] += 1
                else:
                    open_tokens["input"] += input_tokens
                    open_tokens["output"] += output_tokens
                    open_tokens["chats"] += 1
        except Exception as e:
            print(f"[WARN] Error extracting from {path}: {str(e)[:50]}")
            continue
    
    return story_tokens, open_tokens


def _text_value(value):
    """Return readable text from Copilot JSON values."""
    if value is None:
        return ""
    if isinstance(value, str):
        return value
    if isinstance(value, list):
        return "".join(_text_value(item) for item in value)
    if isinstance(value, dict):
        for key in ("value", "text", "content"):
            if key in value:
                return _text_value(value[key])
        return ""
    return str(value)


def _load_session_requests(path):
    """Load Copilot chat requests from JSON or JSONL."""
    if path.endswith(".json"):
        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)
            return data.get("requests", []) if isinstance(data, dict) else []
        except Exception:
            return []

    requests = []
    response_updates = {}
    try:
        with open(path, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                try:
                    entry = json.loads(line)
                except json.JSONDecodeError:
                    continue

                kind = entry.get("kind")
                k = entry.get("k", [])
                v = entry.get("v")

                if kind == 0 and isinstance(v, dict) and "requests" in v:
                    requests = list(v["requests"])
                elif k == ["requests"] and isinstance(v, list):
                    requests.extend(v)
                elif (len(k) == 3 and k[0] == "requests"
                      and isinstance(k[1], int) and k[2] == "response"):
                    response_updates[k[1]] = v if isinstance(v, list) else []
    except Exception:
        return []

    for idx, resp in response_updates.items():
        if idx < len(requests):
            requests[idx]["response"] = resp
    return requests


def get_jira_keys(text):
    """Extract Jira keys from text."""
    matches = re.findall(r"\b[A-Za-z]{2,10}-\d+\b", text)
    return [m.upper() for m in matches if m.upper() not in IGNORE_KEYS]

class CopilotChatEventHandler(FileSystemEventHandler):
    def __init__(self):
        self.last_triggered = {}
    
    def on_modified(self, event):
        if not ((event.src_path.endswith(".json") or event.src_path.endswith(".jsonl"))
                and "chat" in event.src_path.lower()):
            return
        
        try:
            rel_path = os.path.relpath(event.src_path, WORKSPACE_STORAGE)
            workspace_hash = rel_path.split(os.sep)[0]
        except Exception:
            return
        
        now = time.time()
        if now - self.last_triggered.get(workspace_hash, 0) < 2:
            return
        self.last_triggered[workspace_hash] = now
        
        project_name, project_path = get_project_info(workspace_hash)
        print(f"\n[EVENT DETECTED] Chat updated in project: {project_name}")
        
        try:
            process_workspace(workspace_hash)
            print("[SUCCESS] Tokens & costs report updated.")
        except Exception as e:
            print(f"[ERROR] Processing failed: {str(e)[:100]}")
